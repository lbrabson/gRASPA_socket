PATCH_SOCKET_COMMIT_MOVE
  if(SystemComponents.UseSocket)
  {
    // Commit the accepted translation/rotation move to server state.
    // start_position is the atom-level offset; TempVal.molecule is the mol index.
    size_t comp     = SystemComponents.TempVal.component;
    size_t Molsize  = SystemComponents.Moleculesize[comp];
    int    mol_idx  = (int)SystemComponents.TempVal.molecule;
    double3* temp_pos = (double3*) malloc(sizeof(double3) * Molsize);
    cudaMemcpy(temp_pos, Sims.d_a[comp].pos + start_position,
               sizeof(double3) * Molsize, cudaMemcpyDeviceToHost);
    Check_DNNAtom_and_copy_pos_to_UCAtoms(temp_pos, SystemComponents.DNN.UCAtoms[comp],
                                          SystemComponents.ConsiderThisAdsorbateAtom, Molsize);
    SystemComponents.DNN.CommitMove(comp, mol_idx);
    free(temp_pos);
  }

PATCH_SOCKET_COMMIT_INSERT
  if(SystemComponents.UseSocket)
  {
    // After Update_NumberOfMolecules the accepted molecule is at the last slot.
    size_t comp     = SystemComponents.TempVal.component;
    size_t Molsize  = SystemComponents.Moleculesize[comp];
    size_t last_loc = (SystemComponents.NumberOfMolecule_for_Component[comp] - 1) * Molsize;
    double3* temp_pos = (double3*) malloc(sizeof(double3) * Molsize);
    cudaMemcpy(temp_pos, Sims.d_a[comp].pos + last_loc,
               sizeof(double3) * Molsize, cudaMemcpyDeviceToHost);
    Check_DNNAtom_and_copy_pos_to_UCAtoms(temp_pos, SystemComponents.DNN.UCAtoms[comp],
                                          SystemComponents.ConsiderThisAdsorbateAtom, Molsize);
    SystemComponents.DNN.CommitInsert(comp);
    free(temp_pos);
  }

PATCH_SOCKET_COMMIT_DELETE
  if(SystemComponents.UseSocket)
  {
    // TempVal.molecule is the index of the deleted molecule (before swap-with-last).
    size_t comp  = SystemComponents.TempVal.component;
    int mol_idx  = (int)SystemComponents.TempVal.molecule;
    SystemComponents.DNN.CommitDelete(comp, mol_idx);
  }

PATCH_SOCKET_MAIN_PREP
      if(Vars.SystemComponents[a].UseSocket)
      {
        printf("Setting up Socket model\n");
        //Vars.SystemComponents[a].DNN.ReadModel(Vars.SystemComponents[a].ModelName[0]);
        printf("DONE Reading the model, model name %s\n", Vars.SystemComponents[a].ModelName[0].c_str());

        // save pseudo atom symbols but ignore massless sites
        //Vars.SystemComponents[a].DNN.ElementSymbolUsed = Vars.SystemComponents[a].PseudoAtoms.UniqueSymbol;
        std::vector<std::string> filteredSymbols;
        for(const std::string& sym : Vars.SystemComponents[a].PseudoAtoms.UniqueSymbol)
        {
            bool hasMass = false;
            for(size_t i = 0; i < Vars.SystemComponents[a].PseudoAtoms.Symbol.size(); i++)
            {
                if(Vars.SystemComponents[a].PseudoAtoms.Symbol[i] == sym &&
                  Vars.SystemComponents[a].PseudoAtoms.mass[i] > 0.0)
                {
                    hasMass = true;
                    break;
                }
            }
            if(hasMass) filteredSymbols.push_back(sym);
        }
        Vars.SystemComponents[a].DNN.ElementSymbolUsed = filteredSymbols;

        Vars.SystemComponents[a].DNN.Match_Element_PseudoAtom_with_model(Vars.SystemComponents[a].PseudoAtoms);

        Vars.SystemComponents[a].DNN.UCAtoms.resize(Vars.SystemComponents[a].NComponents.x);

        //Copy Boxsize to DNN UCBox//
        Vars.SystemComponents[a].DNN.GenerateUCBox(Vars.Box[a].Cell, Vars.SystemComponents[a].NumberofUnitCells);
        printf("Generated UC Box\n");
        //Copy First unit cell of atoms to UCAtoms//
        //Also Initialize for Adsorbate//
        for(size_t comp = 0; comp < Vars.SystemComponents[a].NComponents.x; comp++)
        {
          int3 Ncells = {1,1,1};
          //If we copy adsorbates, we copy the first template atom (no need to divide number of atoms by unit cells)
          if(comp == 0)
          {
            Ncells = Vars.SystemComponents[a].NumberofUnitCells;
            if(!Vars.SystemComponents[a].rigid[comp] || Vars.SystemComponents[a].NComponents.y != 1)
              throw std::runtime_error("Currently only allows rigid framework and no semi-flexible framework model!!!! If you want, write your own!!!!");
          }
          Vars.SystemComponents[a].DNN.CopyAtomsFromFirstUnitcell(Vars.SystemComponents[a].HostSystem[comp], comp, Ncells, Vars.SystemComponents[a].PseudoAtoms, Vars.SystemComponents[a].ConsiderThisAdsorbateAtom);
        }
        printf("DONE Copying Framework + template adsorbate atom into UCAtoms for Socket\n");
        printf("DONE Setting Test Adsorbate positions\n");

        // Wrap framework and all adsorbate templates into UCBox before connecting
        Vars.SystemComponents[a].DNN.WrapSuperCellAtomIntoUCBox(0);    // framework
        for(size_t comp = 1; comp < Vars.SystemComponents[a].NComponents.x; comp++)
          Vars.SystemComponents[a].DNN.WrapSuperCellAtomIntoUCBox(comp); // adsorbate templates

        // Use box-specific socket path if provided (GRASPA_SOCKET_PATH_0, _1, ...)
        // falling back to GRASPA_SOCKET_PATH for single-box or legacy runs.
        {
          std::string env_key = "GRASPA_SOCKET_PATH_" + std::to_string(a);
          const char* box_path = std::getenv(env_key.c_str());
          if(!box_path) box_path = std::getenv("GRASPA_SOCKET_PATH");
          if(box_path) Vars.SystemComponents[a].DNN.init(box_path, 0);
        }

        // Connect: handshake + species map + framework cache priming
        if(Vars.SystemComponents[a].DNN.connect_socket() < 0)
          throw std::runtime_error("Failed to connect to ML server socket");
        printf("Socket connected and framework cache primed\n");
      }

PATCH_SOCKET_POST_CREATEMOL
    if(Vars.SystemComponents[a].UseSocket)
    {
      // CreateMolecule_InOneBox updates GPU positions only; HostSystem (CPU) is stale.
      // Sync device->host before SyncFull so the server receives the actual accepted positions.
      Atoms device_System[Vars.SystemComponents[a].NComponents.x];
      Copy_AtomData_from_Device(device_System, Vars.SystemComponents[a].HostSystem,
                                Vars.Sims[a].d_a, Vars.SystemComponents[a]);

      // Sync full adsorbate configuration to server after CreateMolecule_InOneBox.
      // Server uses this to prime E_current = E(fw + all_initial_ads).
      for(size_t comp = 1; comp < Vars.SystemComponents[a].NComponents.x; comp++)
      {
        size_t n_mol = Vars.SystemComponents[a].NumberOfMolecule_for_Component[comp];
        if(n_mol == 0) continue;
        Vars.SystemComponents[a].DNN.SyncFull(
            comp,
            n_mol,
            Vars.SystemComponents[a].Moleculesize[comp],
            Vars.SystemComponents[a].HostSystem[comp].pos,
            Vars.SystemComponents[a].ConsiderThisAdsorbateAtom);
      }
      printf("Socket: SyncFull complete after CreateMolecule_InOneBox\n");
    }

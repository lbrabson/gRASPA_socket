void Check_DNNAtom_and_copy_pos_to_UCAtoms(double3* temp_pos, Atoms& UCAtoms, bool* ConsiderThisAdsorbateAtom, size_t Molsize)
{
  size_t update_i = 0;
  for(size_t i = 0; i < Molsize; i++)
  {
    if(ConsiderThisAdsorbateAtom[i])
    {
      UCAtoms.pos[update_i] = temp_pos[i];
      update_i++;
    }
  }
}

PATCH_SOCKET_DELETION
    if(SystemComponents.UseSocket)
    {
      // N-body deletion: QueryDelta returns E(N-1) - E(N) = positive for bound mol.
      // Negate to match classical sign convention (DNN_E = -|binding| = negative).
      int mol_idx = (int)SystemComponents.TempVal.molecule;
      DNN_New = -SystemComponents.DNN.QueryDelta(DELETION, SelectedComponent, mol_idx, SystemComponents.DNNEnergyConversion);
    }

PATCH_SOCKET_SINGLE
    if(SystemComponents.UseSocket)
    {
      if(MoveType == SINGLE_DELETION)
      {
        // QueryDelta returns E(N-1) - E(N) = positive for bound mol.
        // For SINGLE_DELETION: DNN_New - DNN_Old should be positive (cost of removal).
        // Set DNN_New = QueryDelta (positive), DNN_Old = 0.
        int mol_idx = (int)SystemComponents.TempVal.molecule;
        DNN_New = SystemComponents.DNN.QueryDelta(DELETION, SelectedComponent, mol_idx, SystemComponents.DNNEnergyConversion);
        DNN_Old = 0.0;
      }
      else
      {
        // SINGLE_INSERTION, TRANSLATION, ROTATION: need new trial position
        double3* temp_pos = (double3*) malloc(sizeof(double3) * SystemComponents.Moleculesize[SelectedComponent]);
        cudaMemcpy(temp_pos, Sims.New.pos, sizeof(double3) * SystemComponents.Moleculesize[SelectedComponent], cudaMemcpyDeviceToHost);
        Check_DNNAtom_and_copy_pos_to_UCAtoms(temp_pos, SystemComponents.DNN.UCAtoms[SelectedComponent], SystemComponents.ConsiderThisAdsorbateAtom, SystemComponents.Moleculesize[SelectedComponent]);
        int mol_idx = (MoveType == SINGLE_INSERTION) ? -1 : (int)SystemComponents.TempVal.molecule;
        // Use INSERTION for new mol, TRANSLATION (any non-DELETION) for moves
        int query_type = (MoveType == SINGLE_INSERTION) ? INSERTION : TRANSLATION;
        DNN_New = SystemComponents.DNN.QueryDelta(query_type, SelectedComponent, mol_idx, SystemComponents.DNNEnergyConversion);
        DNN_Old = 0.0;
        free(temp_pos);
      }
    }
